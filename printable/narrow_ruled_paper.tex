\documentclass[letterpaper]{article}
\usepackage[margin=0in]{geometry}
\usepackage{tikz}
\pagestyle{empty}

\begin{document}
\begin{tikzpicture}[remember picture, overlay]
    % Colors typical for notebook paper
    \definecolor{rulecolor}{RGB}{130, 180, 220} % Light blue
    \definecolor{margincolor}{RGB}{240, 120, 120} % Light pink/red
    
    % Draw horizontal lines (Narrow ruled: 1/4 inch or 0.25in spacing)
    % Starting from the bottom at 0.5in, going up to 9.75in 
    % (This leaves a 1.25in top header margin on 11in tall paper)
    \foreach \y in {0.5, 0.75, ..., 9.75} {
        \draw[rulecolor, thin] ([yshift=\y in]current page.south west) -- ([yshift=\y in]current page.south east);
    }
    
    % Draw vertical left margin line
    % Typical margin is 1.25in from the left edge
    \draw[margincolor, thick] ([xshift=1.25in]current page.north west) -- ([xshift=1.25in]current page.south west);
    
    % Optional: double margin line (uncomment if you want a double margin)
    % \draw[margincolor, thick] ([xshift=1.29in]current page.north west) -- ([xshift=1.29in]current page.south west);
\end{tikzpicture}
\end{document}
